local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local localPlayer = Players.LocalPlayer
local playerGui = localPlayer:WaitForChild("PlayerGui")

local toggleGui = Instance.new("ScreenGui")
toggleGui.Name = "StealthToggleGui"
toggleGui.ResetOnSpawn = false
toggleGui.Parent = playerGui

local toggleButton = Instance.new("TextButton")
toggleButton.Name = "ToggleButton"
toggleButton.Size = UDim2.new(0, 20, 0, 20)
toggleButton.Position = UDim2.new(0, 20, 0, 100)
toggleButton.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
toggleButton.BackgroundTransparency = 0.7
toggleButton.Text = ""
toggleButton.Parent = toggleGui

local uiCorner = Instance.new("UICorner")
uiCorner.CornerRadius = UDim.new(1, 0)
uiCorner.Parent = toggleButton

local dragging = false
local dragInput, dragStart, startPos
local hasMoved = false

toggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = toggleButton.Position
        hasMoved = false
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

toggleButton.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        if math.abs(delta.X) > 5 or math.abs(delta.Y) > 5 then
            hasMoved = true
        end
        toggleButton.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

local originalVisible = {}
local originalStrokeEnabled = {}

local function hideElement(desc)
    if desc:IsA("GuiObject") then
        if originalVisible[desc] == nil then
            originalVisible[desc] = desc.Visible
        end
        desc.Visible = false
    elseif desc:IsA("UIStroke") then
        if originalStrokeEnabled[desc] == nil then
            originalStrokeEnabled[desc] = desc.Enabled
        end
        desc.Enabled = false
    end
end

local function showElement(desc)
    if desc:IsA("GuiObject") then
        if originalVisible[desc] ~= nil then
            desc.Visible = originalVisible[desc]
        else
            desc.Visible = true
        end
    elseif desc:IsA("UIStroke") then
        if originalStrokeEnabled[desc] ~= nil then
            desc.Enabled = originalStrokeEnabled[desc]
        else
            desc.Enabled = true
        end
    end
end

local isHiding = true

toggleButton.MouseButton1Click:Connect(function()
    if hasMoved then return end
    
    isHiding = not isHiding
    if not isHiding then
        for _, gui in pairs(playerGui:GetChildren()) do
            if gui:IsA("ScreenGui") and gui.Name ~= "StealthToggleGui" then
                for _, desc in pairs(gui:GetDescendants()) do
                    showElement(desc)
                end
            end
        end
        print("_+_++)HOIYB*YB*(^^")
    else
        print("U&{)(BU(U(Y(U")
    end
end)

RunService.RenderStepped:Connect(function()
    if not isHiding then return end
    
    for _, gui in pairs(playerGui:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Name ~= "StealthToggleGui" then
            for _, desc in pairs(gui:GetDescendants()) do
                hideElement(desc)
            end
        end
    end
end)
